""" # Write the HTML to a temporary file html_file_path = "/mnt/data/comprobante_pago.html" with open(html_file_path, "w", encoding="utf-8") as file: file.write(html_content) # Set up headless Chrome options = Options() options.add_argument("--headless") options.add_argument("--no-sandbox") options.add_argument("--disable-dev-shm-usage") options.add_argument("--window-size=800x600") # Capture the screenshot driver = webdriver.Chrome(options=options) driver.get("file://" + html_file_path) screenshot_path = "/mnt/data/comprobante_pago.png" driver.save_screenshot(screenshot_path) driver.quit() # Convert the screenshot to improve resolution image = Image.open(screenshot_path) image = image.resize((800, 600), Image.ANTIALIAS) image.save(screenshot_path) screenshot_path
